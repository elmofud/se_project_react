server {
  server_name api.elmofudfashion.crabdance.com;

  location / {
    proxy_pass http://localhost:3001;
    proxy_http_version 1.1;
    proxy_set_header Upgrade $http_upgrade;
    proxy_set_header Connection 'upgrade';
    proxy_set_header Host $host;
    proxy_cache_bypass $http_upgrade;
  }

  listen 443 ssl;
  ssl_certificate /etc/letsencrypt/live/elmofudfashion.crabdance.com/fullchain.pem;
  ssl_certificate_key /etc/letsencrypt/live/elmofudfashion.crabdance.com/privkey.pem;
  include /etc/letsencrypt/options-ssl-nginx.conf;
  ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;
}
